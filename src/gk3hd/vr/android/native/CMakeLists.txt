cmake_minimum_required(VERSION 3.22.1)
project(gk3hd_android_support LANGUAGES C CXX)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "The standalone native helpers require Android ARM64")
endif()
add_library(gk3hd_pimax SHARED pimax.cpp)
target_compile_features(gk3hd_pimax PRIVATE cxx_std_17)
target_compile_options(gk3hd_pimax PRIVATE -Wall -Wextra -Werror)
target_link_libraries(gk3hd_pimax PRIVATE dl)
add_library(gk3hd_frame_metadata SHARED frame_metadata_reader.c)
target_compile_features(gk3hd_frame_metadata PRIVATE c_std_11)
target_compile_options(gk3hd_frame_metadata PRIVATE -Wall -Wextra -Werror)
add_library(gk3hd_files SHARED durable_files.c)
target_compile_features(gk3hd_files PRIVATE c_std_11)
target_compile_options(gk3hd_files PRIVATE -Wall -Wextra -Werror)
if(DEFINED GK3HD_KGSL_INCLUDE)
    target_sources(gk3hd_files PRIVATE kgsl_allocation_probe.c)
    target_include_directories(gk3hd_files PRIVATE "${GK3HD_KGSL_INCLUDE}")
endif()
