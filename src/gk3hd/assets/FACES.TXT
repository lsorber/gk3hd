//
// Minimum required bitmap names (xxx is model name):
//
//   Face:     xxx_face.bmp
//   Eyelids:  xxx_eyelids.bmp
//   Forehead: xxx_forehead.bmp
//
// Notes:
//
//   * all texture coordinates are always from the upper left corner of
//     the face bitmap.
//   * when referring to texture coordinates, "left" is when looking at the
//     texture (lower values of "x" are to the left).
//   * if any entries are left out, they will be copied from [DEFAULT]. If
//     they do not exist in [DEFAULT] then they will be left blank (or zero).
//   * use [GAB] for documentation purposes
//   * as art is created and moved over, remove the "-XXX" from section names
//     to "uncomment" them. also go into characters.txt and remove the
//     character data from there that relates to texture coordinates and
//     blink frequencies.
//

//////////////////////////////////////////////////////////////////////////////
[Eyes]
//
// Format for eyes: <name>=AxB [,DownSampleOnly]
//
// Where:
//
//   * <name> is the name of the bitmap to use for the eye.
//   * A and B are the number of subpixels to use for the eye. Currently only
//     1x1, 2x2, 4x4 are supported.
//   * DownSampleOnly is an optional flag that says whether or not the source
//     bitmap is already at the higher resolution. If DownSampleOnly is not
//     used, then the source bitmap will be upsampled based on A and B before
//     constructing the subpixel bitmaps.
//
Eye_GreenX   = 4x4, DownSampleOnly
Eye_BrownX   = 4x4, DownSampleOnly
